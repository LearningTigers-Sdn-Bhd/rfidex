// A stand-in for the vendor ECRFID.dll, used only by the native Windows test.
//
// The test builds this file with rustc into a temporary .dll and loads it
// through the real loader, so the ABI, the calling convention, the allocation
// ownership and the freeing paths are all exercised for real. The vendor
// library is never required, and no vendor binary is committed.
//
// The behaviour for one test run is chosen by two environment variables, read
// on every call:
//
//   RFIDEX_FAKE_SCENARIO  which fixture the calls should return
//   RFIDEX_FAKE_LOG       a file every call appends one line to
//
// Everything is allocation-counted: every pointer handed out is registered, and
// freeing a pointer that was never handed out (or handing it out twice) is
// recorded as a fault the test asserts is zero.

use std::collections::HashMap;
use std::ffi::{c_char, c_void};
use std::sync::Mutex;

const UID: [u8; 8] = [0xE0, 0x04, 0x01, 0x50, 0xAB, 0xCD, 0x12, 0x34];
const BLOCKS: usize = 28;
const BLOCK_SIZE: usize = 4;

struct State {
    connect: String,
    /// The sticker's memory, exactly as a desk would read and write it.
    memory: Vec<u8>,
    uid: [u8; 8],
    opened: usize,
    closed: usize,
    /// Pointers handed out and not yet released.
    live: HashMap<usize, usize>,
    /// Enum arrays have pointer-element ownership distinct from byte buffers.
    arrays: HashMap<usize, usize>,
    /// A release of a pointer that was never handed out, or was released twice.
    bad_frees: usize,
    /// Records calls that must never happen at all.
    forbidden: usize,
    last_inventory: (u8, u8),
    last_read: (u8, u8, u8),
    last_write: (u8, u8, Vec<u8>),
    last_records: (u8, u8),
}

impl State {
    fn new() -> State {
        State {
            connect: String::new(),
            memory: Vec::new(),
            uid: UID,
            opened: 0,
            closed: 0,
            live: HashMap::new(),
            arrays: HashMap::new(),
            bad_frees: 0,
            forbidden: 0,
            last_inventory: (0, 0),
            last_read: (0, 0, 0),
            last_write: (0, 0, Vec::new()),
            last_records: (0, 0),
        }
    }
}

static STATE: std::sync::LazyLock<Mutex<State>> = std::sync::LazyLock::new(|| Mutex::new(State::new()));

fn scenario() -> String {
    std::env::var("RFIDEX_FAKE_SCENARIO").unwrap_or_default()
}

fn log(line: &str) {
    let Ok(path) = std::env::var("RFIDEX_FAKE_LOG") else {
        return;
    };
    use std::io::Write;
    if let Ok(mut file) = std::fs::OpenOptions::new()
        .create(true)
        .append(true)
        .open(path)
    {
        let _ = writeln!(file, "{line}");
    }
}

/// Hand out `len` bytes and remember them.
unsafe fn give(len: usize) -> *mut u8 {
    let mut bytes = vec![0u8; len.max(1)];
    let pointer = bytes.as_mut_ptr();
    std::mem::forget(bytes);
    STATE
        .lock()
        .unwrap()
        .live
        .insert(pointer as usize, len.max(1));
    pointer
}

/// Take a pointer back. Returns false if it was never handed out.
unsafe fn take(pointer: *mut u8) -> bool {
    let removed = STATE.lock().unwrap().live.remove(&(pointer as usize));
    match removed {
        Some(len) => {
            drop(Vec::from_raw_parts(pointer, len, len));
            true
        }
        None => {
            STATE.lock().unwrap().bad_frees += 1;
            false
        }
    }
}

unsafe fn give_string(text: &str) -> *mut c_char {
    let bytes = text.as_bytes();
    let pointer = give(bytes.len() + 1);
    std::ptr::copy_nonoverlapping(bytes.as_ptr(), pointer, bytes.len());
    *pointer.add(bytes.len()) = 0;
    pointer as *mut c_char
}

unsafe fn give_array(count: usize) -> *mut *mut u8 {
    let mut pointers = vec![std::ptr::null_mut::<u8>(); count.max(1)].into_boxed_slice();
    let pointer = pointers.as_mut_ptr();
    STATE.lock().unwrap().arrays.insert(pointer as usize, pointers.len());
    std::mem::forget(pointers);
    pointer
}

unsafe fn put(array: *mut *mut u8, index: usize, value: *mut u8) {
    *array.add(index) = value;
}

/// A tag buffer exactly as the vendor demo describes: length, address, command,
/// command, status, DSFID, then the eight UID bytes.
unsafe fn tag_buffer(dsfid: u8, uid: [u8; 8]) -> *mut u8 {
    let mut body = vec![0u8; 15];
    body[0] = 15;
    body[1] = 0x00;
    body[2] = 0xFE;
    body[3] = 0x01;
    body[4] = 0x00;
    body[5] = dsfid;
    body[6..14].copy_from_slice(&uid);
    let pointer = give(body.len());
    std::ptr::copy_nonoverlapping(body.as_ptr(), pointer, body.len());
    pointer
}

unsafe fn terminal_buffer() -> *mut u8 {
    let body = vec![7u8, 0x00, 0xFE, 0x01, 0x00, 0x00, 0x00];
    let size = body[0] as usize;
    let pointer = give(size);
    std::ptr::copy_nonoverlapping(body.as_ptr(), pointer, size);
    pointer
}

#[no_mangle]
pub extern "system" fn FreeBufferArray(array: *mut *mut u8, len: i32) -> i32 {
    unsafe {
        log(&format!("free_buffer_array len={len}"));
        if array.is_null() || len <= 0 {
            STATE.lock().unwrap().bad_frees += 1;
            return -1;
        }
        // Caller owns the outer stack pointer array. Vendor owns only its members.
        for index in 0..len as usize {
            let member = *array.add(index);
            if !member.is_null() && !take(member) {
                return -1;
            }
            *array.add(index) = std::ptr::null_mut();
        }
    }
    0
}

#[no_mangle]
pub extern "system" fn FreeHGlobal(pointer: *mut c_void) -> i32 {
    unsafe {
        log("free_h_global");
        if pointer.is_null() {
            STATE.lock().unwrap().bad_frees += 1;
            return -1;
        }
        let count = STATE.lock().unwrap().arrays.remove(&(pointer as usize));
        if let Some(count) = count {
            drop(Box::from_raw(std::ptr::slice_from_raw_parts_mut(pointer as *mut *mut u8, count)));
            return 0;
        }
        if take(pointer as *mut u8) { 0 } else { -1 }
    }
}

unsafe fn string_list(values: &[&str]) -> *mut *mut c_char {
    let array = give_array(values.len() + 1);
    for (index, value) in values.iter().enumerate() {
        put(array, index, give_string(value) as *mut u8);
    }
    put(array, values.len(), std::ptr::null_mut());
    array as *mut *mut c_char
}

unsafe fn remember(connect: *mut c_char) {
    let text = std::ffi::CStr::from_ptr(connect).to_string_lossy().into_owned();
    STATE.lock().unwrap().connect = text;
}

    model: *mut c_char,
    path: *mut c_char,
    addr_mode: i32,
    exclusive: i32,
) -> *mut c_char {
    let model = unsafe { std::ffi::CStr::from_ptr(model) }.to_string_lossy();

    unsafe {
        let pointer = give_string(&text);
        remember(pointer);
        pointer
    }
}

#[no_mangle]
pub extern "system" fn Open(connect: *mut c_char) -> *mut c_void {
    if connect.is_null() {
        log("open null");
        return std::ptr::null_mut();
    }
    let text = unsafe { std::ffi::CStr::from_ptr(connect) }
        .to_string_lossy()
        .into_owned();
    log(&format!("open {text}"));
    if text.contains("noopen") {
        return std::ptr::null_mut();
    }
    let mut state = STATE.lock().unwrap();
    state.connect = text;
    state.opened += 1;
    state.memory = vec![0u8; BLOCKS * BLOCK_SIZE];
    // A recognisable baseline so a write can be told from a read.
    for (index, byte) in state.memory.iter_mut().enumerate() {
        *byte = (index % 251) as u8;
    }
    0x1 as *mut c_void
}

#[no_mangle]
pub extern "system" fn Close(_ctx: *mut c_void) -> i32 {
    log("close");
    STATE.lock().unwrap().closed += 1;
    0
}

#[no_mangle]
pub extern "system" fn GetDeviceInfo(_ctx: *mut c_void, receive: *mut u8) -> i32 {
    log("get_device_info");
    unsafe {
        std::ptr::write_bytes(receive, 0, 256);
        for (index, byte) in [0xEC, 0x1E, 0x00, 0xFE, 0x00].iter().enumerate() {
            *receive.add(index) = *byte;
        }
    }
    30
}

#[no_mangle]
pub extern "system" fn TagInventory(
    _ctx: *mut c_void,
    array: *mut *mut u8,
    aip: u8,
    mode: u8,
) -> i32 {
    let scenario = scenario();
    {
        let mut state = STATE.lock().unwrap();
        state.last_inventory = (aip, mode);
    }
    log(&format!("tag_inventory aip={aip} mode={mode} scenario={scenario}"));
    if scenario == "error" {
        return -5;
    }
    let count: usize = match scenario.as_str() {
        "none" => 0,
        "two_tags" | "null_member" | "terminal" => 2,
        "five_tags" => 5,
        "status_bad" | "terminal_bad" | "bad_length" | "zero_length" => 1,
        _ => 1,
    };
    if scenario == "over_capacity" {
        // More buffers than the client's array can describe. Nothing is written
        // and nothing is handed out: the client must refuse before walking it.
        return 97;
    }
    if count == 0 {
        return 0;
    }
    unsafe {
        for index in 0..count {
            let member = match (scenario.as_str(), index) {
                ("null_member", 1) => std::ptr::null_mut(),
                ("terminal", 0) => tag_buffer(0x00, UID),
                ("terminal", _) => terminal_buffer(),
                ("terminal_bad", _) => {
                    let pointer = terminal_buffer();
                    *pointer.add(4) = 0x0F;
                    pointer
                }
                ("status_bad", _) => {
                    let pointer = tag_buffer(0x00, UID);
                    *pointer.add(4) = 0x0F;
                    pointer
                }
                ("bad_length", _) => {
                    let pointer = give(15);
                    std::ptr::write_bytes(pointer, 0, 15);
                    *pointer = 3;
                    pointer
                }
                ("zero_length", _) => {
                    let pointer = give(1);
                    *pointer = 0;
                    pointer
                }
                (_, _) => tag_buffer(0x00, UID),
            };
            put(array, index, member);
        }
    }
    count as i32
}

#[no_mangle]
pub extern "system" fn ISO15693_GetTagSystemInfo(
    _ctx: *mut c_void,
    uid: *mut u8,
    receive: *mut u8,
) -> i32 {
    let scenario = scenario();
    log(&format!(
        "get_tag_system_info uid={}",
        hex(unsafe { std::slice::from_raw_parts(uid, 8) })
    ));
    if scenario == "memory_error" {
        return -4;
    }
    if scenario == "memory_zero" {
        unsafe {
            std::ptr::write_bytes(receive, 0, 32);
            *receive.add(4) = 0x00;
        }
        return 20;
    }
    unsafe {
        std::ptr::write_bytes(receive, 0, 32);
        *receive.add(0) = 20;
        *receive.add(4) = 0x00;
        *receive.add(14) = 0x00;
        *receive.add(15) = 0x00;
        // ISO 15693 "minus one" encoding, as a real reader reports it.
        *receive.add(16) = (BLOCKS - 1) as u8;
        *receive.add(17) = if scenario == "memory_other" { 7 } else { (BLOCK_SIZE - 1) as u8 };
        *receive.add(18) = 0x01;
    }
    20
}

#[no_mangle]
pub extern "system" fn ISO15693_ReadMultipleBlocks(
    _ctx: *mut c_void,
    uid: *mut u8,
    read_sec_sta: u8,
    start: u8,
    count: u8,
    receive: *mut u8,
) -> i32 {
    let scenario = scenario();
    let requested = unsafe { std::slice::from_raw_parts(uid, 8) };
    {
        let mut state = STATE.lock().unwrap();
        state.last_read = (start, count, read_sec_sta);
    }
    log(&format!(
        "read_multiple_blocks uid={} read_sec_sta={read_sec_sta} start={start} count={count} scenario={scenario}",
        hex(requested)
    ));
    if scenario == "read_error" {
        return -4;
    }
    // Like the vendor DLL: `count` is zero-based, so count + 1 blocks come back.
    let groups = if scenario == "read_short" {
        count
    } else {
        count + 1
    };
    let uid_out = if scenario == "read_corrupt" {
        [9u8; 8]
    } else {
        let mut copy = [0u8; 8];
        copy.copy_from_slice(requested);
        copy
    };
    let memory = STATE.lock().unwrap().memory.clone();
    let region: Vec<u8> = (0..groups as usize)
        .flat_map(|group| {
            let block = start as usize + group;
            let mut out = vec![0u8];
            let from = block * BLOCK_SIZE;
            let to = (from + BLOCK_SIZE).min(memory.len());
            if from < memory.len() {
                out.extend_from_slice(&memory[from..to]);
            }
            out
        })
        .collect();
    unsafe {
        let length = 17 + region.len();
        std::ptr::write_bytes(receive, 0, 256);
        *receive = length as u8;
        *receive.add(1) = 0x00;
        *receive.add(2) = 0xFE;
        *receive.add(3) = 0x23;
        *receive.add(4) = 0x00;
        std::ptr::copy_nonoverlapping(uid_out.as_ptr(), receive.add(5), 8);
        std::ptr::copy_nonoverlapping(region.as_ptr(), receive.add(15), region.len());
    }
    0
}

#[no_mangle]
pub extern "system" fn ISO15693_WriteMultipleBlocks(
    _ctx: *mut c_void,
    uid: *mut u8,
    start: u8,
    count: u8,
    buffer: *mut u8,
    receive: *mut u8,
) -> i32 {
    let scenario = scenario();
    let data = unsafe { std::slice::from_raw_parts(buffer, count as usize * BLOCK_SIZE).to_vec() };
    {
        let mut state = STATE.lock().unwrap();
        state.last_write = (start, count, data.clone());
    }
    log(&format!(
        "write_multiple_blocks uid={} start={start} count={count} data={} scenario={scenario}",
        hex(unsafe { std::slice::from_raw_parts(uid, 8) }),
        hex(&data)
    ));
    unsafe {
        std::ptr::write_bytes(receive, 0, 8);
    }
    if scenario == "write_error" {
        unsafe {
            *receive.add(4) = 0x0F;
        }
        return 0;
    }
    // A torn write: the sticker leaves the field partway through.
    let written = if scenario == "write_tear" && data.len() > 2 {
        data.len() / 2
    } else {
        data.len()
    };
    let uid_requested = unsafe { std::slice::from_raw_parts(uid, 8) };
    {
        let mut state = STATE.lock().unwrap();
        if state.uid != uid_requested {
            return -1;
        }
        let from = start as usize * BLOCK_SIZE;
        if from + data.len() > state.memory.len() {
            return -1;
        }
        state.memory[from..from + written].copy_from_slice(&data[..written]);
    }
    0
}

#[no_mangle]
pub extern "system" fn MeetingGateTakeRecords(
    _ctx: *mut c_void,
    array: *mut *mut u8,
    flag: u8,
    maxcrp: u8,
) -> i32 {
    let scenario = scenario();
    {
        let mut state = STATE.lock().unwrap();
        state.last_records = (flag, maxcrp);
        if flag != 0 || maxcrp != 1 {
            // Anything else would be a delete, an initialise or a paged read.
            // The test asserts this counter is zero.
            state.forbidden += 1;
        }
    }
    log(&format!(
        "take_records flag={flag} maxcrp={maxcrp} scenario={scenario}"
    ));
    if scenario == "records_empty" {
        return 0;
    }
    if scenario == "records_error" {
        return -7;
    }
    if scenario == "records_many" {
        return 4;
    }
    let body: Vec<u8> = vec![0x0B, 0x00, 0xFE, 0x01, 0x00, 0x01, 0x02, 0x03, 0x04, 0x05, 0x06];
    unsafe {
        let pointer = give(body.len());
        std::ptr::copy_nonoverlapping(body.as_ptr(), pointer, body.len());
        put(array, 0, pointer);
    }
    1
}

#[no_mangle]
pub extern "system" fn FakeConnectString(buffer: *mut c_char, len: i32) -> i32 {
    let text = STATE.lock().unwrap().connect.clone();
    let bytes = text.as_bytes();
    let keep = bytes.len().min(len.saturating_sub(1) as usize);
    unsafe {
        std::ptr::copy_nonoverlapping(bytes.as_ptr(), buffer as *mut u8, keep);
        *buffer.add(keep) = 0;
    }
    keep as i32
}

fn hex(bytes: &[u8]) -> String {
    bytes.iter().map(|byte| format!("{byte:02X}")).collect()
}
